% Each existing content file begins with \section and uses \subsection for
% its internal divisions. Promote those commands when assembling the book.
\let\BookChapter\chapter
\let\BookSection\section
\newcommand{\IncludeChapter}[1]{%
  \begingroup
  % Each content file's top-level \section becomes a labeled book chapter.
  % For example, \IncludeChapter{merge-sort} defines chap:merge-sort.
  \renewcommand{\section}[1]{%
    \BookChapter{##1}%
    \label{chap:#1}%
  }%
  % Promote source-level subsections to book sections. Starred subsections
  % remain unnumbered while receiving a section-level table-of-contents entry.
  \RenewDocumentCommand{\subsection}{s m}{%
    \IfBooleanTF{##1}{%
      \BookSection*{##2}%
      \addcontentsline{toc}{section}{##2}%
    }{\BookSection{##2}}%
  }%
  \import{#1/}{content.tex}%
  \endgroup
}

% A dedicated opening page for each part. The second argument is reserved
% for an introduction or summary beneath the part title.
\newcommand{\PartTitlePage}[2]{%
  \setpartpreamble[u][0.72\textwidth]{%
    \vspace{2\baselineskip}%
    \hrule
    \vspace{1.5\baselineskip}%
    \begin{minipage}{\linewidth}
      \normalfont\normalsize\raggedright
      #2
    \end{minipage}%
  }%
  \part{#1}%
}
